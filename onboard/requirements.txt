# Install on the Raspberry Pi 3 (64-bit Raspberry Pi OS, Bookworm or later).
#
#   sudo apt update
#   sudo apt install -y python3-opencv python3-numpy python3-yaml python3-venv \
#       gstreamer1.0-tools gstreamer1.0-plugins-good gstreamer1.0-plugins-bad gstreamer1.0-libav
#   python3 -m venv --system-site-packages ~/silm-venv
#   ~/silm-venv/bin/pip install -r requirements.txt
#
# Notes:
#  * OpenCV comes from apt (python3-opencv), NOT pip: the apt build has
#    GStreamer, which the camera needs for the Pi's hardware H.264 decoder; pip's
#    opencv-python has no GStreamer and would shadow it. That's why it is not
#    listed below, and why the venv uses --system-site-packages.
#  * Bookworm refuses system-wide `pip install` ("externally managed
#    environment"), hence the venv; systemd/silm-recon.service runs its python.
#  * The TFLite interpreter: the detector accepts any of three backends and
#    picks whichever is installed -- tflite-runtime (classic), ai-edge-litert
#    (its current renamed successor), or full tensorflow. On a modern Pi OS,
#    `pip install ai-edge-litert` is the reliable one; older setups use
#    tflite-runtime (wheels: https://github.com/google-coral/pycoral/releases).
#  * picamera2 is only needed if camera.source == picamera (CSI camera).
#  * For a Coral USB accelerator also install libedgetpu1-std + pycoral.
#  * TRAINING/EXPORT deps (NOT needed on the Pi -- install on your training box):
#      pip install ultralytics litert_torch
#    See training/README.md.
